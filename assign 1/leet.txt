#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, target;
    bool found = false;

    cout << "LeetCode Approach\n";

    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter array elements one by one:\n";
    for (int i = 0; i < n; i++) {
        cout << "Enter element " << i + 1 << ": ";
        cin >> nums[i];
    }

    cout << "Enter target value: ";
    cin >> target;

    int l = 0;
    int r = n - 1;

    while (l <= r) {
        int mid = l + (r - l) / 2;

        if (nums[mid] == target) {
            found = true;
            break;
        }

        if (nums[l] == nums[mid] && nums[mid] == nums[r]) {
            l++;
            r--;
        } else if (nums[l] <= nums[mid]) {
            if (nums[l] <= target && target < nums[mid]) {
                r = mid - 1;
            } else {
                l = mid + 1;
            }
        } else {
            if (nums[mid] < target && target <= nums[r]) {
                l = mid + 1;
            } else {
                r = mid - 1;
            }
        }
    }

    if (found) {
        cout << "Target found: true\n";
    } else {
        cout << "Target not found: false\n";
    }

    return 0;
}
