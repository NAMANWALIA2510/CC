#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n, target;
    bool found = false;

    cout << "Brute Force Approach: Linear Search" << endl;

    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter " << n << " array elements one by one:" << endl;

    for (int i = 0; i < n; i++) {
        cout << "Enter element " << i + 1 << ": ";
        cin >> nums[i];
    }

    cout << "Enter target value: ";
    cin >> target;

    
    for (int i = 0; i < n; i++) {
        if (nums[i] == target) {
            found = true;
            break;
        }
    }

    if (found) {
        cout << "Target " << target << " is present in the array: true" << endl;
    } else {
        cout << "Target " << target << " is not present in the array: false" << endl;
    }

    return 0;
}
