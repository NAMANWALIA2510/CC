#include <iostream>
#include <vector>
using namespace std;

int main() {
    cout << "Experiment 3.3: Number of Islands Problem" << endl;
    cout << "Subject: CC-II (24CSP-339)" << endl;

    int m, n;
    cout << "Enter number of rows: ";
    cin >> m;
    cout << "Enter number of columns: ";
    cin >> n;

    vector<vector<char>> grid(m, vector<char>(n));
    vector<vector<bool>> visited(m, vector<bool>(n, false));

    cout << "Enter the grid (0 or 1):" << endl;
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cin >> grid[i][j];
        }
    }

    int islands = 0;

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == '1' && !visited[i][j]) {
                islands++;

                vector<pair<int, int>> q;
                q.push_back({i, j});
                visited[i][j] = true;

                int k = 0;
                while (k < q.size()) {
                    int r = q[k].first;
                    int c = q[k].second;
                    k++;

                    if (r > 0 && grid[r - 1][c] == '1' && !visited[r - 1][c]) {
                        visited[r - 1][c] = true;
                        q.push_back({r - 1, c});
                    }

                    if (r < m - 1 && grid[r + 1][c] == '1' && !visited[r + 1][c]) {
                        visited[r + 1][c] = true;
                        q.push_back({r + 1, c});
                    }

                    if (c > 0 && grid[r][c - 1] == '1' && !visited[r][c - 1]) {
                        visited[r][c - 1] = true;
                        q.push_back({r, c - 1});
                    }

                    if (c < n - 1 && grid[r][c + 1] == '1' && !visited[r][c + 1]) {
                        visited[r][c + 1] = true;
                        q.push_back({r, c + 1});
                    }
                }
            }
        }
    }

    cout << "Number of Islands: " << islands << endl;

    return 0;
}
