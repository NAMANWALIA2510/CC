#include <iostream>
#include <vector>
#include <functional>
#include <climits>
using namespace std;

int main() {
    int n, amount;

   
    cout << "Experiment 3.2: Coin Change" << endl;
    cout << "Approach: Brute Force Recursion" << endl;

    cout << "Enter number of coin denominations: ";
    cin >> n;

    vector<int> coins(n);

    cout << "Enter coin denominations one by one:" << endl;

    for (int i = 0; i < n; i++) {
        cout << "Enter coin " << i + 1 << ": ";
        cin >> coins[i];
    }

    cout << "Enter target amount: ";
    cin >> amount;

    function<int(int)> solve = [&](int target) {
        if (target == 0) {
            return 0;
        }

        int best = INT_MAX;

        for (int coin : coins) {
            if (coin <= target) {
                int result = solve(target - coin);

                if (result != INT_MAX) {
                    best = min(best, result + 1);
                }
            }
        }

        return best;
    };

    int answer = solve(amount);

    if (answer == INT_MAX) {
        cout << "Minimum coins required: -1" << endl;
    } else {
        cout << "Minimum coins required: " << answer << endl;
    }

    return 0;
}
