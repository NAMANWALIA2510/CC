#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    cout << "Experiment 3.2: Coin Change" << endl;
    cout << "Optimized Approach: Bottom-Up Dynamic Programming" << endl;

    int n, amount;

    cout << "Enter number of coins: ";
    cin >> n;

    vector<int> coins(n);

    cout << "Enter coin denominations: ";
    for (int i = 0; i < n; i++)
        cin >> coins[i];

    cout << "Enter amount: ";
    cin >> amount;

    vector<int> dp(amount + 1, amount + 1);
    dp[0] = 0;

    for (int a = 1; a <= amount; a++) {
        for (int c : coins) {
            if (c <= a)
                dp[a] = min(dp[a], dp[a - c] + 1);
        }
    }

    if (dp[amount] <= amount)
        cout << "Minimum number of coins required: " << dp[amount] << endl;
    else
        cout << "Amount cannot be formed. Output: -1" << endl;

    return 0;
}
