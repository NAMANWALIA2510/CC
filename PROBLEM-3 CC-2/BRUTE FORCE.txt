#include <iostream>
using namespace std;

int main() {
    int n, target;
     cout << "Search Insert Position by : NAMAN WALIA 24BAI70451 \n";

    cout << "Enter the size of the array: ";
    cin >> n;

    int nums[n];

    cout << "Enter " << n << " sorted elements: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << "Enter the target element: ";
    cin >> target;

    for (int i = 0; i < n; i++) {
        if (nums[i] == target) {
            cout << "Target found at index: " << i << endl;
            return 0;
        }

        if (nums[i] > target) {
            cout << "Target not found." << endl;
            cout << "Target should be inserted at index: " << i << endl;
            return 0;
        }
    }

    cout << "Target not found." << endl;
    cout << "Target should be inserted at index: " << n << endl;

    return 0;
}
