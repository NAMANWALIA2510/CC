#include <iostream>
using namespace std;

int main() {
    int n, target;
    
cout << "Optimised Approach to solve search insert position by : NAMAN WALIA 24BAI70451 \n  ";
    cout << "Enter the size of the array: ";
    cin >> n;

    int nums[n];

    cout << "Enter " << n << " sorted elements: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << "Enter the target element: ";
    cin >> target;

    int left = 0;
    int right = n - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (nums[mid] == target) {
            cout << "Target found at index: " << mid << endl;
            return 0;
        }
        else if (nums[mid] < target) {
            left = mid + 1;
        }
        else {
            right = mid - 1;
        }
    }

    cout << "Target not found." << endl;
    cout << "Target should be inserted at index: " << left << endl;

    return 0;
}
