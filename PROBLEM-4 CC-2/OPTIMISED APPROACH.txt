#include <iostream>
using namespace std;

int main() {
    int n, target;
    cout << "Search in Rotated Sorted Array optimised approach (Modi. binary search) APPROACH By :NAMAN WALIA 24BAI70451 \n";

    cout << "Enter the size of the array: ";
    cin >> n;

    int nums[n];

    cout << "Enter the rotated sorted array: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << "Enter the target element: ";
    cin >> target;

    int left = 0;
    int right = n - 1;

    while (left <= right) {
        int mid = left + (right - left) / 2;

        if (nums[mid] == target) {
            cout << "Target found at index: " << mid << endl;
            return 0;
        }

        if (nums[left] <= nums[mid]) {
            if (nums[left] <= target && target < nums[mid]) {
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        } else {
            if (nums[mid] < target && target <= nums[right]) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
    }

    cout << "Target not found in the array." << endl;
    cout << "Returning -1." << endl;

    return 0;
}
