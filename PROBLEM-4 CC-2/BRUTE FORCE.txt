#include <iostream>
using namespace std;

int main() {
    int n, target;
 cout << "Search in Rotated Sorted Array BRUTE FORCE APPROACH By :NAMAN WALIA 24BAI70451 \n";
    cout << "Enter the size of the array: ";
    cin >> n;

    int nums[n];

    cout << "Enter the elements of the array: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

    cout << "Enter the target element: ";
    cin >> target;

    for (int i = 0; i < n; i++) {
        if (nums[i] == target) {
            cout << "Target found at index: " << i << endl;
            return 0;
        }
    }

    cout << "Target not found in the array." << endl;
    cout << "Returning -1." << endl;

    return 0;
}
