#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;

   cout << "\nExperiment 2.1 : Subsets  Iterative (Bitmask) \n";
    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

   
    cout << "Enter " << n << " unique elements: ";
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }

   
    int total = 1 << n;

    cout << "\nAll possible subsets are:\n";

  
    for (int mask = 0; mask < total; mask++) {

        cout << "[";

        for (int i = 0; i < n; i++) {

            if (mask & (1 << i)) {
                cout << nums[i] << " ";
            }
        }

        cout << "]\n";
    }

    
    cout << "\nTotal number of subsets: " << total << endl;

    return 0;
}
