#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;

    cout << "Exp Name: Find the Duplicate No. using Floyd's Tortoise and Hare Approach\n" << endl;
    cout << "Enter n: ";
    cin >> n;

    vector<int> nums(n + 1);

    cout << "Enter " << n + 1 << " array elements one by one:\n" << endl;

    for (int i = 0; i <= n; i++) {
        cout << "Enter element " << i + 1 << ": ";
        cin >> nums[i];
    }

    int slow = nums[0];
    int fast = nums[0];

    do {
        slow = nums[slow];
        fast = nums[nums[fast]];
    } while (slow != fast);

    slow = nums[0];

    while (slow != fast) {
        slow = nums[slow];
        fast = nums[fast];
    }

    cout << "Repeated number: \n" << slow << endl;

    return 0;
}
