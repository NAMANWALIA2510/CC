#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int n, k;
     cout << "THIS IS BRUTE FORCE METHOD BY NAMAN WALIA , 24BAI70451\n";

    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter the array elements: ";
    for(int i = 0; i < n; i++)
    {
        cin >> nums[i];
    }

    cout << "Enter the value of k: ";
    cin >> k;

    bool found = false;

    for(int i = 0; i < n; i++)
    {
        for(int j = i + 1; j < n; j++)
        {
            if(nums[i] == nums[j] && (j - i) <= k)
            {
                found = true;
                break;
            }
        }

        if(found)
        {
            break;
        }
    }

    if(found)
    {
        cout << "Output: True";
    }
    else
    {
        cout << "Output: False";
    }

    return 0;
}
