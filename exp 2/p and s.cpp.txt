#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int n;
    cout << "THIS IS Prefix and Suffix Products METHOD BY NAMAN WALIA , 24BAI70451\n";
    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter the array elements: ";
    for(int i = 0; i < n; i++)
    {
        cin >> nums[i];
    }

    vector<int> prefix(n);
    vector<int> suffix(n);
    vector<int> answer(n);

    prefix[0] = 1;

    for(int i = 1; i < n; i++)
    {
        prefix[i] = prefix[i - 1] * nums[i - 1];
    }

    suffix[n - 1] = 1;

    for(int i = n - 2; i >= 0; i--)
    {
        suffix[i] = suffix[i + 1] * nums[i + 1];
    }

    for(int i = 0; i < n; i++)
    {
        answer[i] = prefix[i] * suffix[i];
    }

    cout << "Product of Array Except Self: ";

    for(int i = 0; i < n; i++)
    {
        cout << answer[i] << " ";
    }

    return 0;
}
