#include <iostream>
#include <vector>
using namespace std;

int main()
{
    int n;
    cout << "THIS IS Prefix + One Variable METHOD BY NAMAN WALIA , 24BAI70451\n";
    cout << "Enter the number of elements: ";
    cin >> n;

    vector<int> nums(n);

    cout << "Enter the array elements: ";
    for(int i = 0; i < n; i++)
    {
        cin >> nums[i];
    }

    vector<int> answer(n);

    answer[0] = 1;

    for(int i = 1; i < n; i++)
    {
        answer[i] = answer[i - 1] * nums[i - 1];
    }

    int suffixProduct = 1;

    for(int i = n - 1; i >= 0; i--)
    {
        answer[i] = answer[i] * suffixProduct;
        suffixProduct = suffixProduct * nums[i];
    }

    cout << "Product of Array Except Self: ";

    for(int i = 0; i < n; i++)
    {
        cout << answer[i] << " ";
    }

    return 0;
}
